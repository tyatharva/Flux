# OpenMPI install

git clone --recursive -b v5.0.x https://github.com/open-mpi/ompi.git
cd ompi
./autogen.pl
./configure \
  CC=amdclang \
  CXX=amdclang++ \
  FC=amdflang \
  --prefix=/opt/openmpi-rocm \
  --with-rocm=/opt/rocm \
  --with-ucx=/opt/ucx-rocm
make -j$(nproc)
make install


# HDF5 install

git clone https://github.com/HDFGroup/hdf5.git
cd hdf5
git checkout hdf5_1.14.6
mkdir build-hdf5-mpi && cd build-hdf5-mpi
CC=mpicc FC=mpifort \
cmake .. \
  -DCMAKE_INSTALL_PREFIX=/opt/hdf5-mpi \
  -DHDF5_ENABLE_PARALLEL=ON \
  -DHDF5_BUILD_FORTRAN=ON \
  -DHDF5_BUILD_EXAMPLES=OFF \
  -DCMAKE_BUILD_TYPE=Release
make -j$(nproc)
make install
